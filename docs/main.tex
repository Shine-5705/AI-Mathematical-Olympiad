\begin{abstract}

% WHAT: One sentence on the problem
Large language models for mathematical reasoning frequently produce solutions
containing symbolic contradictions and code execution errors that go undetected
during standard supervised fine-tuning.

% HOW: Two sentences on the method
We propose Deductive-State Monte Carlo Tree Search (DS-MCTS), a training and
inference framework that integrates symbolic verification (SymPy) and code
execution checking into the MCTS search process. Invalid reasoning branches
are pruned during tree expansion, and only verified solutions are used for
fine-tuning via rejection sampling.

% RESULTS: Two sentences on findings
We evaluate DS-MCTS across three 7B-parameter base models (NuminaMath-7B-TIR,
OpenMath-Nemotron-7B, Qwen2.5-Math-7B-Instruct) on the OpenMathReasoning
benchmark. Our method achieves [X]\% accuracy improvement over SFT baselines,
with the symbolic+code composite verifier pruning [X]\% of erroneous branches,
and the largest gains observed on olympiad-level problems where verification
catches [X]\% more errors than on routine algebra.

% AVAILABILITY
All code, trained adapters, and experimental results are publicly available.

\end{abstract}

\section{Method}

\subsection{Overview}
% Figure 1: Full pipeline diagram
% problem -> MCTS tree -> [verifier checks each node] -> best verified path -> answer
%
% Two-phase approach:
% Phase 1: SFT on OpenMathReasoning (standard baseline)
% Phase 2: MCTS-RL — generate with MCTS, verify, fine-tune on verified solutions

\subsection{Deductive-State MCTS}

% Definition of "deductive state": the accumulated reasoning text at any tree node,
% which includes all prior reasoning steps, code blocks, and intermediate results.
% Unlike standard MCTS for games, our states are variable-length text sequences.

\subsubsection{Tree Structure}
% - Root node: problem statement
% - Each node: accumulated reasoning text (parent text + new continuation)
% - Children: candidate continuations generated by the LLM
% - Node metadata: visit count, value, depth, verification status

\subsubsection{Selection (UCT)}
% - Standard UCT: value/visits + c * sqrt(ln(parent_visits) / visits)
% - c is configurable (we ablate: 0.5, 1.0, 1.41, 2.0)
% - Unvisited nodes get infinite UCT (explored first)

% UCT formula:
% \begin{equation}
% \text{UCT}(n) = \frac{V(n)}{N(n)} + c \sqrt{\frac{\ln N(\text{parent}(n))}{N(n)}}
% \end{equation}

\subsubsection{Expansion with Verification}
% This is the KEY CONTRIBUTION.
% Standard MCTS: expand node, add all children.
% Our MCTS: expand node, VERIFY each candidate, only add valid ones.
%
% Algorithm:
% 1. Generate N candidate continuations from LLM
% 2. For each candidate:
%    a. Run SymbolicVerifier: parse $...$ equations, check for contradictions
%    b. Run CodeExecutionVerifier: extract ```python blocks, execute in sandbox
%    c. If either fails: PRUNE (don't add to tree)
%    d. If both pass: add as child node
%
% This is where the pruning happens. Invalid branches are eliminated EARLY,
% before the tree wastes iterations exploring them.

% Algorithm 1: DS-MCTS pseudocode
% \begin{algorithm}
% \caption{Deductive-State MCTS with Verification}
% \begin{algorithmic}
% \STATE \textbf{Input:} problem $p$, generator $G$, verifier $V$, iterations $T$
% \STATE root $\leftarrow$ Node($p$)
% \FOR{$t = 1$ to $T$}
%   \STATE node $\leftarrow$ SELECT(root) \COMMENT{UCT traversal}
%   \STATE candidates $\leftarrow$ $G$.generate(node.state, $n$)
%   \FOR{each candidate $c$ in candidates}
%     \STATE valid, reason $\leftarrow$ $V$.verify($c$)
%     \IF{valid}
%       \STATE node.children.append(Node(node.state + $c$))
%     \ENDIF
%   \ENDFOR
%   \STATE reward $\leftarrow$ SCORE(node.state)
%   \STATE BACKPROPAGATE(node, reward)
% \ENDFOR
% \STATE \textbf{return} most-visited leaf path
% \end{algorithmic}
% \end{algorithm}

\subsubsection{Scoring Function}
% TIR-aware scoring (not just "has answer"):
% - 1.0: solution contains \boxed{} (complete answer found)
% - 0.5: has ```python + ```output (code was executed, partial progress)
% - 0.3: has ```python (code written but not yet executed)
% - 0.1: reasoning text only (just started)
%
% This guides MCTS toward solutions that actually execute code
% and reach a final answer.

\subsubsection{Backpropagation}
% Standard: propagate reward up to root, increment visit counts.
% The verified-only expansion means high-value paths are more likely
% to be symbolically consistent.

\subsection{Verification Components}

\subsubsection{Symbolic Verifier (SymPy)}
% Parses the reasoning text for:
% 1. $...$ inline LaTeX equations containing "="
%    - Splits on "=", sends both sides to SymPy
%    - If simplify(LHS - RHS) != 0 and is numeric: CONTRADICTION
%    - Example: "we get $x = 5$" then later "$x = 3$"
% 2. \boxed{} answers containing equations
%    - Same check: if the boxed expression is self-contradictory
%
% Handles edge cases:
% - Skips "!=" and "\neq" (intentional inequalities)
% - Skips expressions SymPy can't parse (graceful degradation)
% - Only flags numeric contradictions (symbolic variables are left alone)

\subsubsection{Code Execution Verifier}
% Extracts ```python ... ``` blocks from TIR reasoning.
% Executes each in a subprocess with timeout (default 10s).
% Checks:
% - returncode != 0 → runtime error → PRUNE
% - TimeoutExpired → infinite loop → PRUNE
% - Success → code block is valid
%
% Security: subprocess isolation, no network access, timeout kills.

\subsubsection{Composite Verifier}
% Runs both verifiers sequentially.
% Fails if EITHER fails.
% We ablate: none, symbolic-only, code-only, both.

\subsection{Training Pipeline}

\subsubsection{Phase 1: Supervised Fine-Tuning (SFT)}
% Standard SFT on OpenMathReasoning TIR split.
% Data format: messages = [{system}, {user: problem}, {assistant: solution}]
% Formatted via tokenizer.apply_chat_template()
% LoRA: r=16, alpha=32, targets: q_proj, k_proj, v_proj, o_proj
% This is the BASELINE — what Numina and NVIDIA already demonstrated works.

\subsubsection{Phase 2: MCTS-RL Fine-Tuning}
% OUR CONTRIBUTION.
% 1. Take SFT model as starting point
% 2. For each problem in training set:
%    - Run DS-MCTS to generate multiple solution paths
%    - Verify each with composite verifier
%    - Keep only solutions that: (a) reach \boxed{}, (b) pass verification
% 3. Fine-tune SFT model on verified solutions only
%    - New LoRA adapter (r=8, lower rank for refinement)
%    - Fewer epochs (2) — don't want to overfit to MCTS style
%
% Key insight: SFT trains on ALL solutions (including flawed ones).
% MCTS-RL trains on ONLY verified solutions.
% The model learns to avoid symbolic contradictions and code errors.
